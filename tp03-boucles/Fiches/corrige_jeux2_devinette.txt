public class Devinette {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        Random rand = new Random();

        int nbad = rand.nextInt(100)+1;
        int propo = 0;
        int essais = 0;
        do{
            System.out.println(("Propose un nombre : "));
            propo = sc.nextInt();
            essais++;
            if(propo < nbad){
                System.out.println("Trop petit");
            }else{
                if (propo > nbad){
                    System.out.println("Trop grand");
                }
            }
            int pourcentage = Math.abs(propo-nbad);

            for(int i = 1; i <= 100; i++){
                if(i <= pourcentage){
                    System.out.print("#");
                }else{
                    System.out.print("-");
                }
            }
            System.out.println("");

        }while(propo != nbad);
        System.out.println("Bravo ! Trouvé en " + essais + " coups.");
    }
}
